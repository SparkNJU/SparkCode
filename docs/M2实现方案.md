### Step 2: `src/tools/registry.ts` — 工具注册表（新建）

**职责**：工具注册、schema 汇总、统一执行管道（pre-execute → 执行 → 后处理）。

```typescript
// tools/registry.ts — 工具注册表 + 执行管道

import type { ToolDefinition, ToolResult, ToolRunContext, ToolCallInput } from './types.js'
import type { Agent } from '../core/loop.js'

export class ToolRegistry {
  private defs = new Map<string, ToolDefinition>()

  /** 注册工具，返回 disposer */
  register(def: ToolDefinition): () => void {
    const name = def.schema.function.name
    if (this.defs.has(name)) {
      throw new Error(`[ToolRegistry] duplicate tool: "${name}"`)
    }
    this.defs.set(name, def)
    return () => { this.defs.delete(name) }
  }

  /** 获取所有工具 schema（供 prompt 注入） */
  schemas() {
    return [...this.defs.values()].map(d => d.schema)
  }

  /** 检查是否有工具注册 */
  hasTools(): boolean {
    return this.defs.size > 0
  }

  /** 统一执行管道 */
  async execute(
    input: ToolCallInput,
    ctx: Omit<ToolRunContext, 'callId'>,
  ): Promise<ToolResult> {
    const def = this.defs.get(input.name)
    if (!def) {
      return {
        content: `Error: unknown tool "${input.name}". Available tools: ${[...this.defs.keys()].join(', ')}`,
        isError: true,
      }
    }

    // 构造完整的 ToolRunContext
    const runCtx: ToolRunContext = {
      ...ctx,
      callId: input.id,
    }

    // 执行工具（包裹超时）
    try {
      const result = await withTimeout(
        def.execute(input.args, runCtx),
        60_000,  // 默认 60s 超时
        runCtx.signal,
      )
      return normalizeResult(result)
    } catch (e) {
      return toolErrorResult(e)
    }
  }
}

// ─── 工具函数 ───

function normalizeResult(result: ToolResult): ToolResult {
  // 确保返回值结构完整
  return {
    content: typeof result.content === 'string' ? result.content : String(result.content),
    isError: Boolean(result.isError),
    meta: result.meta,
  }
}

function toolErrorResult(error: unknown): ToolResult {
  if (error instanceof Error) {
    return {
      content: `Error: ${error.message}`,
      isError: true,
      meta: { name: error.name, stack: error.stack },
    }
  }
  return {
    content: `Error: ${String(error)}`,
    isError: true,
  }
}

async function withTimeout<T>(
  promise: Promise<T>,
  ms: number,
  signal?: AbortSignal,
): Promise<T> {
  return new Promise<T>((resolve, reject) => {
    const timer = setTimeout(() => {
      reject(new Error(`Tool execution timed out after ${ms}ms`))
    }, ms)

    // 取消时清除超时定时器
    signal?.addEventListener('abort', () => {
      clearTimeout(timer)
      reject(new DOMException('Aborted', 'AbortError'))
    }, { once: true })

    promise.then(
      (v) => { clearTimeout(timer); resolve(v) },
      (e) => { clearTimeout(timer); reject(e) },
    )
  })
}
```

**设计要点**：
1. **注册/注销**：`register()` 返回 disposer 函数，支持动态卸载
2. **schema 汇总**：`schemas()` 返回所有已注册工具的 JSON Schema，供 prompt 注入
3. **统一执行管道**：`execute()` 是所有工具的唯一入口，包含：
   - 工具查找（unknown tool → 错误结果）
   - 上下文构造
   - 超时包裹（60s 默认）
   - 异常转结构化错误（不抛出，返回 isError 结果）
4. **错误不抛出**：工具失败转为 `{ isError: true }` 结果返回模型，模型可据此自我纠正

### Step 3: `src/tools/bash.ts` — Bash 工具（新建）

**职责**：执行 Shell 命令，返回 stdout/stderr + 退出码。

```typescript
// tools/bash.ts — Shell 执行工具

import { spawn } from 'node:child_process'
import type { ToolDefinition, ToolResult, ToolRunContext } from './types.js'

export const bashTool: ToolDefinition = {
  schema: {
    type: 'function',
    function: {
      name: 'bash',
      description: '在工作目录执行一条 bash 命令，返回 stdout、stderr 与退出码。用于运行测试、构建、安装依赖、git 操作等。',
      parameters: {
        type: 'object',
        properties: {
          command: {
            type: 'string',
            description: '要执行的完整 bash 命令',
          },
          timeout_ms: {
            type: 'integer',
            description: '超时毫秒，默认 60000',
            default: 60000,
          },
        },
        required: ['command'],
        additionalProperties: false,
      },
    },
  },

  async execute(args, ctx): Promise<ToolResult> {
    const command = (args as { command?: string }).command ?? ''
    const timeoutMs = (args as { timeout_ms?: number }).timeout_ms ?? 60_000

    if (!command.trim()) {
      return { content: 'Error: empty command', isError: true }
    }

    return runForeground(command, ctx, timeoutMs)
  },
}

async function runForeground(
  command: string,
  ctx: ToolRunContext,
  timeoutMs: number,
): Promise<ToolResult> {
  // Windows 兼容：使用 cmd /c 或 powershell
  const isWin = process.platform === 'win32'
  const shell = isWin ? 'cmd' : '/bin/bash'
  const shellArgs = isWin ? ['/c', command] : ['-lc', command]

  const proc = spawn(shell, shellArgs, {
    cwd: ctx.cwd,
    env: { ...process.env, SPARK_CWD: ctx.cwd },
    stdio: ['pipe', 'pipe', 'pipe'],
    timeout: timeoutMs,
  })

  let stdout = ''
  let stderr = ''

  proc.stdout.on('data', (data: Buffer) => {
    stdout += data.toString()
  })

  proc.stderr.on('data', (data: Buffer) => {
    stderr += data.toString()
  })

  // 取消支持：abort → kill 进程
  const onAbort = () => {
    try {
      proc.kill('SIGKILL')
    } catch {
      // 进程可能已退出
    }
  }
  ctx.signal.addEventListener('abort', onAbort, { once: true })

  // 等待进程结束
  const exitCode: number | null = await new Promise((resolve) => {
    proc.on('close', (code) => resolve(code))
    proc.on('error', () => resolve(null))
  })

  ctx.signal.removeEventListener('abort', onAbort)

  // 渲染输出
  const text = renderShellOutput(exitCode, stdout, stderr, ctx.cwd)
  const isError = exitCode !== 0 && exitCode !== null

  // 截断过长输出
  return {
    content: truncate(text, 20_000),
    isError,
    meta: { exitCode, stdoutLen: stdout.length, stderrLen: stderr.length },
  }
}

function renderShellOutput(
  exitCode: number | null,
  stdout: string,
  stderr: string,
  cwd: string,
): string {
  const parts: string[] = []

  if (stdout) {
    parts.push(stdout)
  }
  if (stderr) {
    if (parts.length) parts.push('\n--- stderr ---\n')
    parts.push(stderr)
  }

  parts.push(`\n[exit code: ${exitCode ?? 'null'}]`)

  return parts.join('')
}

function truncate(text: string, max: number): string {
  if (text.length <= max) return text
  const head = text.slice(0, Math.floor(max * 0.8))
  const tail = text.slice(-Math.floor(max * 0.2))
  return `${head}\n\n…[输出已截断，共 ${text.length} 字符]…\n\n${tail}`
}
```

**设计要点**：
1. **跨平台**：Windows 用 `cmd /c`，Unix 用 `/bin/bash -lc`
2. **输出捕获**：分别捕获 stdout 和 stderr，合并渲染
3. **取消支持**：AbortSignal → `proc.kill('SIGKILL')`
4. **输出截断**：超过 20,000 字符时保留头 80% + 尾 20%
5. **退出码**：非零退出码标记为 `isError: true`，模型可读取并调整命令
6. **M1 简化**：不实现后台任务（`run_in_background`），不实现 job 工具族

### Step 4: `src/core/llm.ts` — Usage 提取

**修改点**：启用 `stream_options: { include_usage: true }`，从流中提取 usage。

```typescript
// core/llm.ts — M2 修改点（仅列出变更部分）

// streamModel 函数修改：添加 stream_options
export async function* streamModel(
  client: OpenAI,
  request: GenerateOptions,
  signal: AbortSignal,
): AsyncIterable<StreamChunk> {
  const hasTools = request.tools && request.tools.length > 0

  const stream = await client.chat.completions.create(
    {
      model: request.model,
      messages: request.messages as ChatCompletionMessageParam[],
      stream: true,
      stream_options: { include_usage: true },  // 🆕 启用 usage 输出
      ...(hasTools ? {
        tools: request.tools as OpenAI.ChatCompletionTool[],
        tool_choice: 'auto' as const,
      } : {}),
    },
    { signal },
  )

  for await (const part of stream) {
    // 🆕 提取 usage（在最后一个 chunk 中）
    if (part.usage) {
      yield {
        kind: 'finish',
        reason: 'usage',
        usage: {
          promptTokens: part.usage.prompt_tokens,
          completionTokens: part.usage.completion_tokens,
          totalTokens: part.usage.total_tokens,
        },
      }
    }

    const delta = part.choices[0]?.delta
    if (!delta) continue

    // ... 其余逻辑不变 ...
  }
}

// StreamChunk 类型需要扩展
export type StreamChunk =
  | { kind: 'content'; text: string }
  | { kind: 'tool-call-part'; index: number; id?: string; name?: string; argsFragment: string }
  | { kind: 'finish'; reason: string | null; usage?: TokenUsage }  // 🆕 添加 usage 字段

// LlmAdapter.streamWithRetry 修改：提取 usage
private async streamWithRetry(...) {
  try {
    const assembler = new BlockAssembler()
    let usage: TokenUsage | undefined

    for await (const chunk of streamModel(this.client, request, signal)) {
      assembler.feed(chunk)
      // 🆕 提取 usage
      if (chunk.kind === 'finish' && chunk.usage) {
        usage = chunk.usage
      }
      // 实时广播（不变）
      this.ctx.emit('assistant/chunk', { ... })
    }

    // ... 其余逻辑不变，但 return 中 usage 已有值 ...
    return { message, usage }
  }
}
```

**要点**：
- `stream_options: { include_usage: true}` 让 OpenAI 在最后一个 chunk 返回 usage
- usage 通过 `finish` chunk 传递给调用方
- `assistant/message` 事件中 now 包含 usage 数据

export interface PromptAssembly {

#### 6.3 runTurn() 重构（核心）

```typescript
/** 运行一个回合：支持 multi-step 工具调用 */
private async runTurn(): Promise<boolean> {
  this.turn++
  this.step = 0
  this.abort = new AbortController()

  this.session.append('turn/start', { turn: this.turn })

  try {
    // (A) 认领 next-turn 输入
    const messages = this.inbox.claim('next-turn', this.turn)
    if (messages.length === 0) {
      this.session.append('turn/end', { turn: this.turn, reason: { kind: 'completed' } })
      return false
    }

    // 记录用户消息
    this.session.append('step/start', { turn: this.turn, step: this.step })
    for (const m of messages) {
      this.session.append('user/message', m, { surfaceOp: 'append' })
    }

    // (B) 主循环：LLM 请求 → tool-call → 执行 → 循环
    while (true) {
      // 组装 prompt（通过 waterfall，支持工具 schema 注入）
      const tools = this.tools.hasTools() ? this.tools.schemas() : undefined
      const assembly = await this.ctx.waterfall<PromptAssembly>(
        'prompt/assemble',
        assemblePrompt(this.session, this.config, tools),
      )

      // 构建历史 + API 消息
      const history = this.session.deriveMessages()
      const apiMessages = [
        { role: 'system' as const, content: assembly.header.systemPrompt },
        ...toApiMessages(history),
      ]
      const request: GenerateOptions = {
        model: assembly.header.model,
        messages: apiMessages,
        tools: assembly.header.tools,  // 🆕 传工具 schema
      }

      // 记录请求头
      this.session.append('request/header', {
        header: { model: assembly.header.model, systemPrompt: assembly.header.systemPrompt },
        reason: 'change',
      })

      // 调用 LLM（流式）
      this.llm.setCurrentTurn(this.turn, this.step)
      const result = await this.llm.stream(request, this.abort.signal)

      // 记录 assistant 消息
      this.session.append('assistant/message', {
        turn: this.turn,
        step: this.step,
        message: result.message,
        usage: result.usage,
      }, { surfaceOp: 'append' })

      // 提取 tool-call
      const toolCalls = result.message.content
        .filter((b): b is Extract<ContentBlock, { type: 'tool-call' }> => b.type === 'tool-call')

      // 无 tool-call → 回合完成
      if (toolCalls.length === 0) {
        this.session.append('step/end', { turn: this.turn, step: this.step })
        break
      }

      // 🆕 执行工具调用
      await this.executeToolCalls(toolCalls)

      this.session.append('step/end', { turn: this.turn, step: this.step })

      // 检查 step 上限
      if (this.step >= this.config.maxStepsPerTurn) {
        break
      }

      // 准备下一步：认领 next-step 输入
      this.step++
      const nextMessages = this.inbox.claim('next-step', this.turn)
      if (nextMessages.length > 0) {
        // 有新的用户输入，记录后继续
        this.session.append('step/start', { turn: this.turn, step: this.step })
        for (const m of nextMessages) {
          this.session.append('user/message', m, { surfaceOp: 'append' })
        }
      } else {
        // 无新输入，继续循环（工具结果已在历史中）
        this.session.append('step/start', { turn: this.turn, step: this.step })
      }
    }

    // 回合结束
    this.session.append('turn/end', { turn: this.turn, reason: { kind: 'completed' } })
    return this.inbox.hasPending()
  } catch (error) {
    const failure = toLlmFailure(error)
    this.session.append('turn/end', {
      turn: this.turn,
      reason: { kind: 'error', error: failure },
    })
    throw error
  }
}
```

#### 6.4 新增 executeToolCalls 方法

```typescript
/** 执行工具调用（按模型顺序） */
private async executeToolCalls(
  calls: Array<{ id: string; name: string; arguments: string }>,
): Promise<void> {
  // 记录所有 tool/call 事件
  for (const call of calls) {
    this.session.append('tool/call', {
      turn: this.turn,
      step: this.step,
      callId: call.id,
      name: call.name,
      arguments: call.arguments,
    })
  }

  // 逐个执行（串行，简单可靠）
  for (const call of calls) {
    let args: Record<string, unknown>
    try {
      args = call.arguments ? JSON.parse(call.arguments) : {}
    } catch {
      args = { _raw: call.arguments }
    }

    const result = await this.tools.execute(
      { id: call.id, name: call.name, args },
      {
        signal: this.abort.signal,
        agent: this,
        cwd: this.cwd,
        deferContext: (msg) => this.inject(msg),
        writeEvent: (type, data) => this.session.append(type as any, data as any),
      },
    )

    // 记录 tool/result 事件
    this.session.append('tool/result', {
      turn: this.turn,
      step: this.step,
      message: {
        id: generateId(),
        role: 'tool',
        callId: call.id,
        content: result.content,
        isError: result.isError,
      },
      error: result.isError ? { name: 'ToolError', code: 'TOOL_FAILED' } : undefined,
    }, { surfaceOp: 'append' })
  }
}
```

## 5. 命令模式（`!` 前缀）

### 5.1 功能描述

REPL 中支持两种输入模式：

| 前缀 | 模式 | 行为 | 消耗 token |
|---|---|---|---|
| `!` | 命令模式 | 直接执行 Shell 命令，不经过 LLM | ❌ |
| 无 | 对话模式 | 发送给 LLM，Agent 决定是否调用工具 | ✅ |

示例：
```
> !ls -la           ← 直接执行，打印结果
> !pwd              ← 直接执行
> 帮我写个 Hello World  ← 走 LLM 对话
> 运行 npm test     ← 走 LLM，LLM 决定调用 bash 工具
```

### 5.2 实现（index.ts REPL 修改）

```typescript
// REPL 输入循环中：
const trimmed = input.trim()

// 命令模式：! 前缀 → 直接执行
if (trimmed.startsWith('!')) {
  const command = trimmed.slice(1).trim()
  if (!command) continue
  await executeDirectCommand(agent.tools, command, config)
  continue
}

// 对话模式：走 LLM
agent.followup(trimmed)
await agent.waitForTurnEnd()
```

```typescript
/** 直接执行 Shell 命令（不经过 LLM） */
async function executeDirectCommand(
  tools: ToolRegistry,
  command: string,
  config: SparkConfig,
): Promise<void> {
  const color = config.noColor ? '' : '\x1b[36m'
  const reset = config.noColor ? '' : '\x1b[0m'
  stdout.write(`${color}$ ${command}${reset}\n`)

  const result = await tools.execute(
    { id: 'direct', name: 'bash', args: { command } },
    {
      signal: AbortSignal.timeout(60_000),
      agent: null as any,  // 命令模式不需要 agent
      cwd: config.workspace,
      deferContext: () => {},
      writeEvent: () => {},
    },
  )

  if (result.isError) {
    const errColor = config.noColor ? '' : '\x1b[31m'
    stdout.write(`${errColor}${result.content}${reset}\n`)
  } else {
    stdout.write(`${result.content}\n`)
  }
}
```

## 6. 关键设计决策（答辩准备）

### Q: 工具失败为什么不抛异常，而是返回 isError 结果？

**A**：抛异常会终止整个回合，用户只能重新开始。而返回 `{ isError: true }` 结果，模型能读到错误原因并**自主修正**（换命令、修路径、重试）。这是"智能体自主完成任务"的核心机制——只有回合级致命错误才抛出。

### Q: 工具执行为什么是串行而非并行？

**A**：M2 采用串行执行（`for...of` + `await`），原因：(1) 简单可靠，易于调试；(2) bash 命令可能有依赖关系（前一个命令的输出影响下一个）；(3) 并行执行需要更复杂的资源管理（并发限制、输出交错）。后续可开放并行，但**结果必须按模型顺序提交**。

### Q: prompt 组装为什么用 waterfall 而不是直接拼接？

**A**：waterfall 中间件链让工具 schema 注入、上下文压缩等横切逻辑与核心循环解耦。M2 只有一个 handler（工具注入），但 M4 加压缩时只需注册新 handler，不修改 `loop.ts`。这借鉴了 DSH 的"类型化事件注入策略"。

### Q: 为什么 bash 工具不实现后台运行？

**A**：后台任务（`run_in_background`）需要 JobRegistry + job_output/job_kill/job_list 工具族，实现复杂度高。M2 聚焦核心路径——前台执行 + 超时 + 取消。后台任务放到 M6（TUI 完善）实现，那时交互体验更成熟。

### Q: 输出截断为什么保留头 80% + 尾 20%？

**A**：头部通常包含命令输出的开头（标题、前几行结果），尾部通常包含最终结果或错误信息。中间部分（如大量日志）对模型理解结果帮助最小。截断后提示"完整输出见落盘文件"，模型可决定是否需要读取完整内容。
